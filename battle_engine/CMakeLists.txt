cmake_minimum_required(VERSION 3.19)
project(battle_engine LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ENGINE "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/sts_lightspeed")
file(GLOB_RECURSE ENGINE_SOURCES CONFIGURE_DEPENDS "${ENGINE}/src/*.cpp")
add_library(sts_engine STATIC ${ENGINE_SOURCES})
target_include_directories(sts_engine PUBLIC "${ENGINE}/include" "${ENGINE}/json/include")
add_library(battle_core STATIC src/Snapshot.cpp src/SelectionSnapshot.cpp src/TurnSearch.cpp src/MonsterRoster.cpp)
target_link_libraries(battle_core PUBLIC sts_engine)
add_executable(battle-engine-worker src/main.cpp)
target_link_libraries(battle-engine-worker PRIVATE battle_core)
target_compile_definitions(battle-engine-worker PRIVATE "$<$<CONFIG:Release>:AIP_RELEASE_BUILD>")
target_compile_definitions(battle_core PUBLIC NOMINMAX)
set(BATTLE_ENGINE_OUTPUT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/bin" CACHE PATH "Worker output directory")
set_target_properties(battle-engine-worker PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${BATTLE_ENGINE_OUTPUT_DIR}")
if(MSVC)
  target_compile_options(sts_engine PRIVATE /FI"${CMAKE_CURRENT_SOURCE_DIR}/src/MsvcCompat.h" /EHsc /utf-8)
  target_compile_options(battle_core PUBLIC /FI"${CMAKE_CURRENT_SOURCE_DIR}/src/MsvcCompat.h" /EHsc /utf-8)
endif()

if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "BattleAiMod needs a 64-bit toolchain")
  endif()
  find_path(BATTLE_JNI_INCLUDE jni.h HINTS "$ENV{JAVA_HOME}/include" REQUIRED)
  add_library(battle_jni SHARED src/NativeBindings.cpp)
  target_include_directories(battle_jni PRIVATE "${BATTLE_JNI_INCLUDE}" "${BATTLE_JNI_INCLUDE}/win32")
  target_link_libraries(battle_jni PRIVATE battle_core)
  set_target_properties(battle_jni PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${BATTLE_ENGINE_OUTPUT_DIR}")
endif()
